import type { MotionValue } from 'framer-motion'
import { motion } from 'framer-motion'

interface LayerTransforms {
  y: MotionValue<number>
  scale: MotionValue<number>
  rotateX: MotionValue<number>
}

interface Label3DModelProps {
  front: LayerTransforms & { boxShadow: MotionValue<string> }
  middle: {
    y: MotionValue<number>
    opacity: MotionValue<number>
  }
  back: LayerTransforms & { blur: MotionValue<number> }
}

const LAYER_STYLES = {
  back: {
    container:
      'absolute flex h-full w-full items-center justify-center rounded-2xl border-2 bg-gradient-to-br from-brand-blue/5 to-brand-blue/10 border-brand-blue/20 shadow-lg',
    label: 'text-lg font-medium tracking-widest uppercase text-brand-blue/40',
    text: 'Base',
  },
  middle: {
    container:
      'absolute -inset-6 flex items-center justify-center rounded-2xl border bg-gradient-to-br from-slate-100 to-slate-200 border-slate-300 shadow-xl',
    label: 'text-xs font-medium tracking-[0.2em] uppercase text-slate-500',
    text: 'Satén Premium',
  },
  front: {
    container:
      'relative z-10 flex h-full w-full flex-col items-center justify-center gap-4 rounded-2xl border-2 bg-white border-brand-blue text-brand-blue',
    label: 'text-lg font-bold tracking-wider uppercase',
    subtitle: 'text-sm font-medium',
  },
} as const

function BackLayer({ transforms }: { transforms: Label3DModelProps['back'] }) {
  return (
    <motion.div
      style={{ y: transforms.y, scale: transforms.scale, rotateX: transforms.rotateX, filter: transforms.blur }}
      className={LAYER_STYLES.back.container}
    >
      <span className={LAYER_STYLES.back.label}>{LAYER_STYLES.back.text}</span>
    </motion.div>
  )
}

function MiddleLayer({ transforms }: { transforms: Label3DModelProps['middle'] }) {
  return (
    <motion.div
      style={{ y: transforms.y, opacity: transforms.opacity }}
      className={LAYER_STYLES.middle.container}
    >
      <div className="flex flex-col items-center gap-2">
        <svg width="48" height="48" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1" className="text-slate-400">
          <path d="M4 6l16-2M4 12l16-2M4 18l16-2" />
        </svg>
        <span className={LAYER_STYLES.middle.label}>{LAYER_STYLES.middle.text}</span>
      </div>
    </motion.div>
  )
}

function FrontLayer({ transforms }: { transforms: Label3DModelProps['front'] }) {
  return (
    <motion.div
      style={{ y: transforms.y, scale: transforms.scale, rotateX: transforms.rotateX, boxShadow: transforms.boxShadow }}
      className={LAYER_STYLES.front.container}
    >
      <svg width="64" height="64" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" className="text-brand-blue">

      </svg>
      <span className={LAYER_STYLES.front.label}>JL</span>
      <span className={LAYER_STYLES.front.subtitle}>Jonalabels</span>
    </motion.div>
  )
}

export default function Label3DModel({ front, middle, back }: Label3DModelProps) {
  return (
    <div
      className="relative flex h-80 w-72 items-center justify-center md:h-96 md:w-80"

import { useRef } from 'react'
import { useNavigate } from 'react-router-dom'
import { useScroll, useTransform, useSpring, motion } from 'framer-motion'
import Label3DModel from './Label3DModel'

const SPRING = { damping: 20, stiffness: 90 }

export default function LandingPage() {
  const navigate = useNavigate()
  const containerRef = useRef<HTMLDivElement>(null)

  const { scrollYProgress } = useScroll({
    target: containerRef,
    offset: ['start start', 'end end'],
  })

  const raw = useTransform(scrollYProgress, [0, 1], [0, 1])
  const smooth = useSpring(raw, SPRING)

  const frontY = useTransform(smooth, [0, 1], [0, -120])
  const frontScale = useTransform(smooth, [0, 1], [1, 1.08])
  const frontRotateX = useTransform(smooth, [0, 1], [0, -8])
  const frontBoxShadow = useTransform(
    smooth, [0, 1],
    ['0 8px 32px rgba(27,42,74,0.15)', '0 16px 48px rgba(27,42,74,0.25)'],
  )

  const middleY = useTransform(smooth, [0, 0.5, 1], [0, -30, 0])
  const middleOpacity = useTransform(smooth, [0, 0.3, 0.7, 1], [0, 1, 1, 0.6])

  const backY = useTransform(smooth, [0, 1], [0, 120])
  const backScale = useTransform(smooth, [0, 1], [1, 0.92])
  const backRotateX = useTransform(smooth, [0, 1], [0, 8])
  const backBlur = useTransform(smooth, [0, 1], [0, 4])

  const heroOpacity = useTransform(smooth, [0, 0.3], [1, 0])
  const heroScale = useTransform(smooth, [0, 0.3], [1, 0.95])
  const headlineY = useTransform(smooth, [0, 0.25], [0, -60])
  const headlineOpacity = useTransform(smooth, [0, 0.2], [1, 0])

  const ctaOpacity = useTransform(smooth, [0.6, 0.8, 1], [0, 1, 1])
  const ctaY = useTransform(smooth, [0.6, 0.8, 1], [40, 0, 0])

  const handleCtaClick = () => {
    navigate('/cotizar')
  }

  return (
    <section ref={containerRef} className="relative" aria-label="Sección principal de bienvenida">
      <div className="h-[300vh]" />

      <div className="fixed inset-0 flex items-center justify-center bg-brand-light">
        <motion.div
          style={{ opacity: heroOpacity, scale: heroScale }}
          className="absolute inset-0 flex flex-col items-center justify-center"
        >
          <motion.h1
            style={{ y: headlineY, opacity: headlineOpacity }}
            className="max-w-3xl px-6 text-center text-4xl font-bold leading-tight tracking-tight text-brand-blue md:text-5xl lg:text-6xl"
          >
            Etiquetas que dan identidad
          </motion.h1>
        </motion.div>

        <Label3DModel
          front={{ y: frontY, scale: frontScale, rotateX: frontRotateX, boxShadow: frontBoxShadow }}
          middle={{ y: middleY, opacity: middleOpacity }}
          back={{ y: backY, scale: backScale, rotateX: backRotateX, blur: backBlur }}
        />
      </div>

      <motion.div
        style={{ opacity: ctaOpacity, y: ctaY }}
        className="relative z-20 flex flex-col items-center justify-center px-6 pb-32"
      >
        <p className="mb-8 max-w-xl text-center text-lg text-brand-blue/70">
          Dale identidad a tus prendas con etiquetas de satén premium personalizadas.
        </p>
        <motion.button
          onClick={handleCtaClick}
          whileHover={{ scale: 1.05 }}
          whileTap={{ scale: 0.95 }}
          className="cursor-pointer rounded-xl bg-brand-orange px-10 py-4 text-lg font-semibold text-white shadow-lg shadow-brand-orange/30"
        >
          Cotizar mi diseño
        </motion.button>
      </motion.div>
    </section>
  )
}

5.9 src/features/pedido/pedidoService.ts
────────────────────────────────────────────────────────────────────────────────
import http from '../../lib/axios'

const MATERIAL_PRODUCT_MAP: Record<string, number> = {
  SATEN: 1,
  DAMASCO: 2,
  ALGODON: 3,
}

export interface IPedidoResponse {
  id: number
  usuarioId: number
  productoId: number
  disenoId: number
  tallerId: number | null
  estado: string
  cantidad: number
  precioFinalCotizado: number | null
  costoTallerAcordado: number | null
  comentariosAdmin: string | null
  urlDiseno: string | null
  fechaCreacion: string
  fechaActualizacion: string
}

export function getProductoIdByMaterial(tipo: string): number {
  const id = MATERIAL_PRODUCT_MAP[tipo]
  if (!id) {
    throw new Error(`Tipo de material no válido: ${tipo}`)
  }
  return id
}

export async function crearPedido(formData: FormData): Promise<IPedidoResponse> {
  const { data } = await http.post<IPedidoResponse>('/pedidos', formData, {
    headers: { 'Content-Type': 'multipart/form-data' },
  })
  return data
}

5.10 src/features/pedido/QuoteForm.tsx
────────────────────────────────────────────────────────────────────────────────
import { type FormEvent, useState } from 'react'
import { crearPedido, getProductoIdByMaterial } from './pedidoService'
import type { IPedidoResponse } from './pedidoService'

type MaterialTipo = 'SATEN' | 'DAMASCO' | 'ALGODON'

interface IQuoteFormState {
  tipoMaterial: MaterialTipo | ''
  ancho: string
  alto: string
  cantidad: string
}

interface ISubmitState {
  loading: boolean
  error: string | null
  success: boolean
  pedido: IPedidoResponse | null
}

const INPUT_CLASS =
  'w-full rounded-lg border border-gray-300 bg-white px-4 py-3 text-sm text-gray-900 outline-none transition focus:border-brand-blue focus:ring-2 focus:ring-brand-blue/20'
const LABEL_CLASS = 'mb-1.5 block text-sm font-medium text-brand-blue'

const MATERIAL_OPTIONS: { value: MaterialTipo; label: string }[] = [
  { value: 'SATEN', label: 'Satén' },
  { value: 'DAMASCO', label: 'Damasco' },
  { value: 'ALGODON', label: 'Algodón' },
]

export default function QuoteForm() {
  const [form, setForm] = useState<IQuoteFormState>({
    tipoMaterial: '', ancho: '', alto: '', cantidad: '',
  })
  const [file, setFile] = useState<File | null>(null)
  const [submit, setSubmit] = useState<ISubmitState>({
    loading: false, error: null, success: false, pedido: null,
  })

  const updateField = (name: keyof IQuoteFormState, value: string) =>
    setForm((prev) => ({ ...prev, [name]: value }))

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) =>
    setFile(e.target.files?.[0] ?? null)

  const validate = (): string | null => {
    if (!form.tipoMaterial) return 'Selecciona un tipo de material'
    if (!form.ancho || Number(form.ancho) <= 0) return 'Indica un ancho válido'
    if (!form.alto || Number(form.alto) <= 0) return 'Indica un alto válido'
    if (!form.cantidad || Number(form.cantidad) <= 0) return 'Indica una cantidad válida'
    if (!file) return 'Sube el archivo de tu diseño o logo'
    return null
  }

  const resetForm = () => {
    setForm({ tipoMaterial: '', ancho: '', alto: '', cantidad: '' })
    setFile(null)
    setSubmit({ loading: false, error: null, success: false, pedido: null })
  }

  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault()
    setSubmit({ loading: false, error: null, success: false, pedido: null })

    const errorMsg = validate()
    if (errorMsg) {
      setSubmit({ loading: false, error: errorMsg, success: false, pedido: null })
      return
    }

    const formData = new FormData()
    formData.append('archivo', file!)
    formData.append('tipoMaterial', form.tipoMaterial)
    formData.append('ancho', form.ancho)
    formData.append('alto', form.alto)
    formData.append('cantidad', form.cantidad)
    formData.append('usuarioId', '1')
    formData.append('productoId', String(getProductoIdByMaterial(form.tipoMaterial as MaterialTipo)))
    formData.append('disenoId', '1')

    setSubmit({ loading: true, error: null, success: false, pedido: null })

    try {
      const pedido = await crearPedido(formData)
      setSubmit({ loading: false, error: null, success: true, pedido })
    } catch (err: unknown) {
      const message = err instanceof Error ? err.message : 'Ocurrió un error inesperado'
      setSubmit({ loading: false, error: message, success: false, pedido: null })
    }
  }

  // Success screen
  if (submit.success && submit.pedido) {
    return (
      <section className="mx-auto max-w-lg px-4 py-16 text-center" aria-label="Pedido creado exitosamente">
        <div className="mb-4 text-5xl">✓</div>
        <h2 className="mb-2 text-2xl font-bold text-brand-blue">Solicitud enviada</h2>
        <p className="mb-6 text-brand-blue/70">
          Tu pedido <span className="font-semibold">#{submit.pedido.id}</span> está en
          revisión. Te contactaremos pronto.
        </p>
        <button onClick={resetForm}
          className="cursor-pointer rounded-lg bg-brand-orange px-8 py-3 font-semibold text-white shadow-lg shadow-brand-orange/30 transition hover:scale-105 active:scale-95">
          Nueva cotización
        </button>
      </section>
    )
  }

  // Form
  return (
    <section className="mx-auto w-full max-w-xl px-4 py-12" aria-label="Formulario de cotización">
      <h2 className="mb-8 text-3xl font-bold text-brand-blue">Cotiza tus etiquetas</h2>

      <form onSubmit={handleSubmit} className="space-y-6" noValidate>
        {/* Tipo de material */}
        <div>
          <label htmlFor="tipoMaterial" className={LABEL_CLASS}>Tipo de material</label>
          <select id="tipoMaterial" value={form.tipoMaterial}
            onChange={(e) => updateField('tipoMaterial', e.target.value)}
            className={INPUT_CLASS}>
            <option value="">Selecciona un material</option>
            {MATERIAL_OPTIONS.map((opt) => (
              <option key={opt.value} value={opt.value}>{opt.label}</option>
            ))}
          </select>
        </div>

        {/* Dimensiones */}
        <div className="grid grid-cols-2 gap-4">
          <div>
            <label htmlFor="ancho" className={LABEL_CLASS}>Ancho (cm)</label>
            <input id="ancho" type="number" min={1} step={0.1}
              value={form.ancho} onChange={(e) => updateField('ancho', e.target.value)}
              className={INPUT_CLASS} placeholder="Ej: 5" />
          </div>
          <div>
            <label htmlFor="alto" className={LABEL_CLASS}>Alto (cm)</label>
            <input id="alto" type="number" min={1} step={0.1}
              value={form.alto} onChange={(e) => updateField('alto', e.target.value)}
              className={INPUT_CLASS} placeholder="Ej: 3" />
          </div>
        </div>

        {/* Cantidad */}
        <div>
          <label htmlFor="cantidad" className={LABEL_CLASS}>Cantidad</label>
          <input id="cantidad" type="number" min={1}
            value={form.cantidad} onChange={(e) => updateField('cantidad', e.target.value)}
            className={INPUT_CLASS} placeholder="Ej: 500" />
        </div>

        {/* Archivo */}
        <div>
          <label htmlFor="archivo" className={LABEL_CLASS}>Logo o diseño</label>
          <input id="archivo" type="file" accept=".jpg,.jpeg,.png,.pdf" onChange={handleFileChange}
            className={`${INPUT_CLASS} file:mr-4 file:cursor-pointer file:rounded-lg file:border-0 file:bg-brand-orange/10 file:px-4 file:py-2 file:text-sm file:font-medium file:text-brand-orange hover:file:bg-brand-orange/20`} />
          {file && <p className="mt-1.5 text-xs text-brand-blue/60">{file.name} ({(file.size / 1024).toFixed(1)} KB)</p>}
        </div>

        {/* Error message */}
        {submit.error && (
          <div className="rounded-lg border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
            {submit.error}
          </div>
        )}

        {/* Submit button */}
        <button type="submit" disabled={submit.loading}
          className="w-full cursor-pointer rounded-xl bg-brand-orange px-10 py-4 text-lg font-semibold text-white shadow-lg shadow-brand-orange/30 transition hover:scale-[1.02] active:scale-[0.98] disabled:cursor-not-allowed disabled:opacity-60">
          {submit.loading ? 'Procesando...' : 'Enviar cotización'}
        </button>
      </form>
    </section>
  )
}

7. PROXY DE DESARROLLO
--------------------------------------------------------------------------------
Todas las peticiones a /api/* se redirigen a http://localhost:8080 (backend Java)
gracias al proxy configurado en vite.config.ts.

8. FLUJO DE DATOS (QuoteForm → Backend)
--------------------------------------------------------------------------------
1. Usuario llena el formulario (material, dimensiones, cantidad, archivo)
2. Al hacer submit, se construye un FormData con todos los campos
3. POST /api/v1/pedidos con Content-Type: multipart/form-data
4. Las cookies HttpOnly viajan con withCredentials: true
5. Si la respuesta es exitosa (201), se muestra pantalla de confirmación
6. Si falla, se muestra el mensaje de error en la UI
7. El material se mapea a productoId: SATEN→1, DAMASCO→2, ALGODON→3

9. SCROLLYTELLING (LandingPage)
--------------------------------------------------------------------------------
- useScroll + useSpring (damping:20, stiffness:90) para animación suave
- 3 capas de etiqueta que se separan en el eje Z al hacer scroll:
  * Frontal: sube (y -120), escala (1.08), rotaX (-8°)
  * Media: aparece con fade, se desplaza ligeramente
  * Trasera: baja (y 120), se reduce (0.92), blur (4px)
- Hero headline se desvanece en el primer 30% del scroll
- CTA aparece al 60-80% del scroll
- 300vh de espacio para permitir la animación completa
